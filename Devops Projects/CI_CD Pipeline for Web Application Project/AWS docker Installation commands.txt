        ECS_TASK_DEFINITION = 'my-task-definition'
    }
    stages {
        stage('Checkout Code') {
            steps {
                // Fetch code from your SCM (e.g., Git)
                git branch: 'main', credentialsId: 'github-credential-id', url: 'github.com'
            }
        }
        stage('Build Image') {
            steps {
                script {
		    sh "mvn package"
                    sh "docker build -t $ECR_REGISTRY/$ECR_REPO:$IMAGE_TAG ."
                }
            }
        }
        stage('Push to ECR') {
            steps {
                // Authenticate Docker with ECR and push the image

                }
            }
        }
        stage('Deploy to ECS') {
            steps {
                script {
                    // This uses AWS CLI to update the ECS service

                }
            }
